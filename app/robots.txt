import { MetadataRoute } from 'next';

export default function robots(): MetadataRoute.Robots {
  return {
    rules: [
      {
        userAgent: '*',
        allow: '/',
        disallow: ['/api/', '/dashboard/', '/authenticated/', '/p/'],
      },
    ],
    sitemap: 'https://erp.bababuilds.com/sitemap.xml',
  };
}
